## Overview

Implement a comprehensive file-based logging system with rotation, log levels, and structured output for license release service operations, audit trails, and troubleshooting.

## Technical Approach

### 1. File-Based Logging Framework
- Create thread-safe file logging with proper synchronization
- Implement log file rotation based on size and time
- Support for multiple log files (service, audit, error)
- Include log formatting and structured output

### 2. Log Level Management
- Implement standard log levels (Debug, Info, Warning, Error, Critical)
- Configurable log level filtering
- Support for different log levels per component
- Performance-optimized logging with conditional compilation

### 3. File Rotation Strategy
- Size-based rotation with configurable limits
- Time-based rotation (daily, weekly, monthly)
- Automatic cleanup of old log files
- Compression support for archived logs

### 4. Logging Configuration
- Configuration-based log settings
- Support for different environments (development, production)
- Runtime log level changes
- Detailed logging control for troubleshooting

### Logging Framework
```csharp
public interface ILoggerService
{
    void LogDebug(string message, params object[] args);
    void LogInfo(string message, params object[] args);
    void LogWarning(string message, params object[] args);
    void LogError(string message, Exception exception = null);
    void LogCritical(string message, Exception exception = null);
    void LogAudit(string action, string details, string user = null);
    void LogMetric(string metricName, double value, Dictionary<string, string> tags = null);
}

public enum LogLevel
{
    Debug,
    Info,
    Warning,
    Error,
    Critical
}

public class FileLoggerService : ILoggerService
{
    private readonly LoggingOptions _options;
    private readonly object _lock = new object();
    private StreamWriter _currentLogWriter;
    private StreamWriter _auditLogWriter;
    private StreamWriter _errorLogWriter;
    private string _currentLogFileName;
    private string _auditLogFileName;
    private string _errorLogFileName;
    private long _currentLogSize;
    private DateTime _lastRotationCheck;

    public FileLoggerService(LoggingOptions options)
    {
        _options = options;
        InitializeLoggers();
        SetupRotationTimer();
    }

    private void InitializeLoggers()
    {
        try
        {
            var logDirectory = Path.GetDirectoryName(_options.LogFilePath);
            if (!Directory.Exists(logDirectory))
            {
                Directory.CreateDirectory(logDirectory);
            }

            // Initialize main log file
            _currentLogFileName = GenerateLogFileName(_options.LogFilePath);
            _currentLogWriter = new StreamWriter(_currentLogFileName, true)
            {
                AutoFlush = true
            };

            // Initialize audit log file
            var auditLogPath = Path.Combine(logDirectory, "audit.log");
            _auditLogFileName = GenerateLogFileName(auditLogPath);
            _auditLogWriter = new StreamWriter(_auditLogFileName, true)
            {
                AutoFlush = true
            };

            // Initialize error log file
            var errorLogPath = Path.Combine(logDirectory, "error.log");
            _errorLogFileName = GenerateLogFileName(errorLogPath);
            _errorLogWriter = new StreamWriter(_errorLogFileName, true)
            {
                AutoFlush = true
            };

            _currentLogSize = new FileInfo(_currentLogFileName).Length;
            _lastRotationCheck = DateTime.Now;

            LogInfo("Logging system initialized successfully");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Failed to initialize logging system: {ex.Message}");
            throw;
        }
    }

    public void LogDebug(string message, params object[] args)
    {
        if (_options.MinimumLevel <= LogLevel.Debug)
        {
            WriteLog(LogLevel.Debug, message, args);
        }
    }

    public void LogInfo(string message, params object[] args)
    {
        if (_options.MinimumLevel <= LogLevel.Info)
        {
            WriteLog(LogLevel.Info, message, args);
        }
    }

    public void LogWarning(string message, params object[] args)
    {
        if (_options.MinimumLevel <= LogLevel.Warning)
        {
            WriteLog(LogLevel.Warning, message, args);
        }
    }

    public void LogError(string message, Exception exception = null)
    {
        WriteLog(LogLevel.Error, message, null, exception);

        // Also write to error log file
        WriteToErrorLog(LogLevel.Error, message, exception);
    }

    public void LogCritical(string message, Exception exception = null)
    {
        WriteLog(LogLevel.Critical, message, null, exception);

        // Also write to error log file
        WriteToErrorLog(LogLevel.Critical, message, exception);
    }

    public void LogAudit(string action, string details, string user = null)
    {
        var auditMessage = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] ACTION: {action} | DETAILS: {details} | USER: {user ?? "system"}";

        lock (_lock)
        {
            _auditLogWriter.WriteLine(auditMessage);
            _auditLogWriter.Flush();
        }
    }

    public void LogMetric(string metricName, double value, Dictionary<string, string> tags = null)
    {
        var tagString = tags != null ? string.Join(",", tags.Select(t => $"{t.Key}={t.Value}")) : "";
        var metricMessage = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] METRIC: {metricName}={value} | TAGS: {tagString}";

        WriteLog(LogLevel.Info, metricMessage);
    }

    private void WriteLog(LogLevel level, string message, object[] args = null, Exception exception = null)
    {
        var formattedMessage = args != null ? string.Format(message, args) : message;
        var logEntry = FormatLogEntry(level, formattedMessage, exception);

        lock (_lock)
        {
            _currentLogWriter.WriteLine(logEntry);
            _currentLogWriter.Flush();

            _currentLogSize += logEntry.Length + Environment.NewLine.Length;

            CheckRotationNeeded();
        }
    }

    private void WriteToErrorLog(LogLevel level, string message, Exception exception = null)
    {
        var formattedMessage = exception != null ? $"{message} | EXCEPTION: {exception.Message} | STACK: {exception.StackTrace}" : message;
        var errorEntry = FormatLogEntry(level, formattedMessage, exception);

        lock (_lock)
        {
            _errorLogWriter.WriteLine(errorEntry);
            _errorLogWriter.Flush();
        }
    }

    private string FormatLogEntry(LogLevel level, string message, Exception exception = null)
    {
        var timestamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff");
        var levelString = level.ToString().ToUpper();
        var exceptionInfo = exception != null ? $" | EXCEPTION: {exception.Message} | STACK: {exception.StackTrace}" : "";

        return $"[{timestamp}] [{levelString}] {message}{exceptionInfo}";
    }

    private void CheckRotationNeeded()
    {
        var now = DateTime.Now;

        // Check rotation every minute
        if (now - _lastRotationCheck < TimeSpan.FromMinutes(1))
        {
            return;
        }

        _lastRotationCheck = now;

        // Size-based rotation
        if (_currentLogSize >= _options.MaxFileSizeBytes)
        {
            RotateLogFiles();
            return;
        }

        // Time-based rotation
        if (_options.RotationInterval.HasValue)
        {
            var fileInfo = new FileInfo(_currentLogFileName);
            if (now - fileInfo.CreationTime > _options.RotationInterval.Value)
            {
                RotateLogFiles();
            }
        }
    }

    private void RotateLogFiles()
    {
        lock (_lock)
        {
            try
            {
                // Close current writers
                _currentLogWriter?.Dispose();
                _auditLogWriter?.Dispose();
                _errorLogWriter?.Dispose();

                // Rename current files with timestamp
                var timestamp = DateTime.Now.ToString("yyyyMMdd_HHmmss");

                if (File.Exists(_currentLogFileName))
                {
                    var archiveName = Path.ChangeExtension(_currentLogFileName, $".{timestamp}.log");
                    File.Move(_currentLogFileName, archiveName);

                    // Compress if enabled
                    if (_options.EnableCompression)
                    {
                        CompressFile(archiveName);
                    }
                }

                // Create new log files
                _currentLogFileName = GenerateLogFileName(_options.LogFilePath);
                _currentLogWriter = new StreamWriter(_currentLogFileName, true)
                {
                    AutoFlush = true
                };

                _auditLogFileName = GenerateLogFileName(Path.Combine(Path.GetDirectoryName(_options.LogFilePath), "audit.log"));
                _auditLogWriter = new StreamWriter(_auditLogFileName, true)
                {
                    AutoFlush = true
                };

                _errorLogFileName = GenerateLogFileName(Path.Combine(Path.GetDirectoryName(_options.LogFilePath), "error.log"));
                _errorLogWriter = new StreamWriter(_errorLogFileName, true)
                {
                    AutoFlush = true
                };

                _currentLogSize = 0;

                // Clean up old files
                CleanupOldFiles();

                LogInfo("Log files rotated successfully");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Failed to rotate log files: {ex.Message}");
            }
        }
    }

    private string GenerateLogFileName(string basePath)
    {
        var directory = Path.GetDirectoryName(basePath);
        var fileName = Path.GetFileNameWithoutExtension(basePath);
        var extension = Path.GetExtension(basePath);
        var date = DateTime.Now.ToString("yyyyMMdd");

        return Path.Combine(directory, $"{fileName}_{date}{extension}");
    }

    private void CompressFile(string filePath)
    {
        try
        {
            var compressedPath = filePath + ".gz";
            using (var originalStream = File.OpenRead(filePath))
            using (var compressedStream = File.Create(compressedPath))
            using (var compressionStream = new GZipStream(compressedStream, CompressionMode.Compress))
            {
                originalStream.CopyTo(compressionStream);
            }

            File.Delete(filePath);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Failed to compress file {filePath}: {ex.Message}");
        }
    }

    private void CleanupOldFiles()
    {
        if (!_options.MaxArchiveDays.HasValue)
        {
            return;
        }

        try
        {
            var logDirectory = Path.GetDirectoryName(_options.LogFilePath);
            var cutoffDate = DateTime.Now.AddDays(-_options.MaxArchiveDays.Value);

            var files = Directory.GetFiles(logDirectory, "*.log*", SearchOption.TopDirectoryOnly)
                .Concat(Directory.GetFiles(logDirectory, "*.gz", SearchOption.TopDirectoryOnly));

            foreach (var file in files)
            {
                var fileInfo = new FileInfo(file);
                if (fileInfo.CreationTime < cutoffDate)
                {
                    try
                    {
                        File.Delete(file);
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"Failed to delete old log file {file}: {ex.Message}");
                    }
                }
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Failed to cleanup old log files: {ex.Message}");
        }
    }

    private void SetupRotationTimer()
    {
        var timer = new Timer(RotationTimerCallback, null, TimeSpan.FromMinutes(1), TimeSpan.FromMinutes(1));
    }

    private void RotationTimerCallback(object state)
    {
        CheckRotationNeeded();
    }

    public void Dispose()
    {
        lock (_lock)
        {
            _currentLogWriter?.Dispose();
            _auditLogWriter?.Dispose();
            _errorLogWriter?.Dispose();
        }
    }
}
```

### Logging Configuration
```csharp
public class LoggingOptions
{
    public string LogFilePath { get; set; } = "logs/license-service.log";
    public LogLevel MinimumLevel { get; set; } = LogLevel.Info;
    public long MaxFileSizeBytes { get; set; } = 10 * 1024 * 1024; // 10MB
    public TimeSpan? RotationInterval { get; set; } = TimeSpan.FromDays(1);
    public int? MaxArchiveDays { get; set; } = 30;
    public bool EnableCompression { get; set; } = true;
    public bool EnableAuditLogging { get; set; } = true;
    public bool EnableErrorLogging { get; set; } = true;
    public bool EnableMetricsLogging { get; set; } = true;
}

public class LoggingConfigurationProvider
{
    private readonly IConfiguration _configuration;

    public LoggingConfigurationProvider(IConfiguration configuration)
    {
        _configuration = configuration;
    }

    public LoggingOptions GetLoggingOptions()
    {
        return new LoggingOptions
        {
            LogFilePath = _configuration["Logging:LogFilePath"] ?? "logs/license-service.log",
            MinimumLevel = Enum.Parse<LogLevel>(_configuration["Logging:MinimumLevel"] ?? "Info"),
            MaxFileSizeBytes = long.Parse(_configuration["Logging:MaxFileSizeBytes"] ?? "10485760"),
            RotationInterval = _configuration["Logging:RotationInterval"] != null
                ? TimeSpan.Parse(_configuration["Logging:RotationInterval"])
                : (TimeSpan?)TimeSpan.FromDays(1),
            MaxArchiveDays = _configuration["Logging:MaxArchiveDays"] != null
                ? int.Parse(_configuration["Logging:MaxArchiveDays"])
                : (int?)30,
            EnableCompression = bool.Parse(_configuration["Logging:EnableCompression"] ?? "true"),
            EnableAuditLogging = bool.Parse(_configuration["Logging:EnableAuditLogging"] ?? "true"),
            EnableErrorLogging = bool.Parse(_configuration["Logging:EnableErrorLogging"] ?? "true"),
            EnableMetricsLogging = bool.Parse(_configuration["Logging:EnableMetricsLogging"] ?? "true")
        };
    }
}
```

### Structured Logging Support
```csharp
public class StructuredLogger
{
    private readonly ILoggerService _logger;
    private readonly LoggingOptions _options;

    public StructuredLogger(ILoggerService logger, LoggingOptions options)
    {
        _logger = logger;
        _options = options;
    }

    public void LogStructuredEvent(string eventType, Dictionary<string, object> properties)
    {
        var json = JsonSerializer.Serialize(properties);
        _logger.LogInfo($"STRUCTURED_EVENT: {eventType} | {json}");
    }

    public void LogLicenseEvent(string action, LicenseInfo license, Dictionary<string, object> additionalProperties = null)
    {
        var properties = new Dictionary<string, object>
        {
            ["Action"] = action,
            ["Feature"] = license.Feature,
            ["UserHost"] = license.UserHost,
            ["ClientHost"] = license.ClientHost,
            ["BorrowTime"] = license.BorrowTime,
            ["UsageDuration"] = license.UsageDuration,
            ["Status"] = license.Status
        };

        if (additionalProperties != null)
        {
            foreach (var prop in additionalProperties)
            {
                properties[prop.Key] = prop.Value;
            }
        }

        LogStructuredEvent("LicenseEvent", properties);
    }

    public void LogServiceEvent(string action, Dictionary<string, object> properties)
    {
        properties["Action"] = action;
        properties["Timestamp"] = DateTime.Now;
        properties["Service"] = "LicenseReleaseService";

        LogStructuredEvent("ServiceEvent", properties);
    }
}
```
