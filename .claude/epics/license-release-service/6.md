## Overview

Implement a robust timer-based execution system using System.Timers.Timer for periodic license checks, ensuring reliable and configurable scheduling of license monitoring and release operations.

## Technical Approach

### 1. Timer Framework Architecture
- Create configurable timer wrapper with proper lifecycle management
- Implement timer event handling with error recovery
- Support for dynamic timer interval adjustments
- Timer synchronization and thread safety

### 2. Execution Scheduling
- Implement configurable check intervals
- Support for different execution modes (continuous, scheduled)
- Handle timer overlaps and concurrent execution prevention
- Implement graceful shutdown and cancellation support

### 3. Error Handling and Recovery
- Robust error handling for timer events
- Automatic recovery from timer failures
- Implement circuit breaker pattern for repeated failures
- Proper disposal and cleanup of timer resources

### 4. Performance Optimization
- Minimize timer overhead and resource usage
- Implement efficient timer recycling
- Support for high-frequency checks without performance degradation
- Memory management for long-running operations

public interface ITimerExecutor
{
    void Start(TimeSpan interval);
    void Stop();
    void Pause();
    void Resume();
    void UpdateInterval(TimeSpan newInterval);
    bool IsRunning { get; }
    TimerStatus Status { get; }
    event EventHandler<TimerExecutionEventArgs> ExecutionStarted;
    event EventHandler<TimerExecutionEventArgs> ExecutionCompleted;
    event EventHandler<TimerErrorEventArgs> ExecutionError;
}

public class TimerExecutor : ITimerExecutor, IDisposable
{
    private readonly ILogger<TimerExecutor> _logger;
    private readonly TimerExecutionOptions _options;
    private Timer _timer;
    private readonly object _lock = new object();
    private bool _isRunning;
    private bool _isPaused;
    private bool _isExecuting;
    private int _consecutiveErrors;
    private DateTime _lastExecutionTime;
    private TimerStatus _status;

    public event EventHandler<TimerExecutionEventArgs> ExecutionStarted;
    public event EventHandler<TimerExecutionEventArgs> ExecutionCompleted;
    public event EventHandler<TimerErrorEventArgs> ExecutionError;

    public TimerExecutor(ILogger<TimerExecutor> logger, TimerExecutionOptions options)
    {
        _logger = logger;
        _options = options;
        _status = TimerStatus.Stopped;
    }

    public void Start(TimeSpan interval)
    {
        lock (_lock)
        {
            if (_isRunning)
            {
                _logger.LogWarning("Timer is already running");
                return;
            }

            _timer = new Timer(interval.TotalMilliseconds);
            _timer.Elapsed += OnTimerElapsed;
            _timer.AutoReset = true;
            _timer.Enabled = true;

            _isRunning = true;
            _isPaused = false;
            _status = TimerStatus.Running;
            _consecutiveErrors = 0;

            _logger.LogInformation("Timer started with interval {Interval}ms", interval.TotalMilliseconds);
        }
    }

    public void Stop()
    {
        lock (_lock)
        {
            if (!_isRunning)
            {
                return;
            }

            try
            {
                _timer?.Stop();
                _timer?.Dispose();
                _timer = null;

                _isRunning = false;
                _isPaused = false;
                _status = TimerStatus.Stopped;

                _logger.LogInformation("Timer stopped");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error stopping timer");
                throw;
            }
        }
    }

    public void Pause()
    {
        lock (_lock)
        {
            if (!_isRunning || _isPaused)
            {
                return;
            }

            _timer?.Stop();
            _isPaused = true;
            _status = TimerStatus.Paused;

            _logger.LogInformation("Timer paused");
        }
    }

    public void Resume()
    {
        lock (_lock)
        {
            if (!_isRunning || !_isPaused)
            {
                return;
            }

            _timer?.Start();
            _isPaused = false;
            _status = TimerStatus.Running;

            _logger.LogInformation("Timer resumed");
        }
    }

    public void UpdateInterval(TimeSpan newInterval)
    {
        lock (_lock)
        {
            if (!_isRunning)
            {
                _logger.LogWarning("Timer is not running, cannot update interval");
                return;
            }

            _timer?.Stop();
            _timer = new Timer(newInterval.TotalMilliseconds);
            _timer.Elapsed += OnTimerElapsed;
            _timer.AutoReset = true;
            _timer.Enabled = true;

            _logger.LogInformation("Timer interval updated to {Interval}ms", newInterval.TotalMilliseconds);
        }
    }

    public bool IsRunning => _isRunning;
    public TimerStatus Status => _status;

    private async void OnTimerElapsed(object sender, ElapsedEventArgs e)
    {
        if (_isExecuting)
        {
            _logger.LogWarning("Timer execution already in progress, skipping");
            return;
        }

        lock (_lock)
        {
            if (_isExecuting || _isPaused || !_isRunning)
            {
                return;
            }

            _isExecuting = true;
        }

        try
        {
            var startTime = DateTime.Now;
            var args = new TimerExecutionEventArgs
            {
                ExecutionId = Guid.NewGuid(),
                StartTime = startTime,
                Attempt = 1
            };

            ExecutionStarted?.Invoke(this, args);

            await ExecuteOperationAsync(args);

            var endTime = DateTime.Now;
            args.EndTime = endTime;
            args.Duration = endTime - startTime;

            ExecutionCompleted?.Invoke(this, args);

            _lastExecutionTime = endTime;
            _consecutiveErrors = 0;

            _logger.LogDebug("Timer execution completed in {Duration}ms", args.Duration.TotalMilliseconds);
        }
        catch (Exception ex)
        {
            await HandleExecutionErrorAsync(ex);
        }
        finally
        {
            lock (_lock)
            {
                _isExecuting = false;
            }
        }
    }

    private async Task ExecuteOperationAsync(TimerExecutionEventArgs args)
    {
        try
        {
            _logger.LogDebug("Starting timer execution {ExecutionId}", args.ExecutionId);

            // Execute the main operation
            await ExecuteLicenseCheckAsync();

            _logger.LogDebug("Timer execution {ExecutionId} completed successfully", args.ExecutionId);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Timer execution {ExecutionId} failed", args.ExecutionId);
            throw;
        }
    }

    private async Task ExecuteLicenseCheckAsync()
    {
        // This will be implemented to coordinate with license management
        _logger.LogDebug("Executing license check operation");

        // Placeholder for actual license checking logic
        await Task.Delay(100); // Simulate work

        _logger.LogDebug("License check operation completed");
    }

    private async Task HandleExecutionErrorAsync(Exception ex)
    {
        _consecutiveErrors++;

        var errorArgs = new TimerErrorEventArgs
        {
            Error = ex,
            ConsecutiveErrors = _consecutiveErrors,
            Timestamp = DateTime.Now
        };

        ExecutionError?.Invoke(this, errorArgs);

        _logger.LogError(ex, "Timer execution failed (consecutive errors: {Count})", _consecutiveErrors);

        // Implement circuit breaker pattern
        if (_consecutiveErrors >= _options.MaxConsecutiveErrors)
        {
            await HandleCircuitBreakerAsync();
        }
    }

    private async Task HandleCircuitBreakerAsync()
    {
        _logger.LogWarning("Circuit breaker triggered due to {Count} consecutive errors", _consecutiveErrors);

        // Stop the timer to prevent further failures
        Stop();

        // Wait for cooldown period
        await Task.Delay(_options.CircuitBreakerCooldown);

        // Reset error count and attempt to restart
        _consecutiveErrors = 0;

        try
        {
            Start(TimeSpan.FromMilliseconds(_options.DefaultIntervalMs));
            _logger.LogInformation("Timer restarted after circuit breaker cooldown");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to restart timer after circuit breaker cooldown");
        }
    }

    public void Dispose()
    {
        Dispose(true);
        GC.SuppressFinalize(this);
    }

    protected virtual void Dispose(bool disposing)
    {
        if (disposing)
        {
            Stop();
        }
    }
}

public interface ILicenseCheckScheduler
{
    void Start();
    void Stop();
    void Pause();
    void Resume();
    void UpdateSchedule(TimeSpan newInterval);
    SchedulerStatus Status { get; }
    event EventHandler<LicenseCheckEventArgs> LicenseCheckStarted;
    event EventHandler<LicenseCheckEventArgs> LicenseCheckCompleted;
}

public class LicenseCheckScheduler : ILicenseCheckScheduler
{
    private readonly ITimerExecutor _timerExecutor;
    private readonly ILogger<LicenseCheckScheduler> _logger;
    private readonly ILicenseManager _licenseManager;
    private readonly ILicenseQueryEngine _queryEngine;
    private readonly TimerExecutionOptions _options;
    private SchedulerStatus _status;

    public event EventHandler<LicenseCheckEventArgs> LicenseCheckStarted;
    public event EventHandler<LicenseCheckEventArgs> LicenseCheckCompleted;

    public LicenseCheckScheduler(ITimerExecutor timerExecutor, ILogger<LicenseCheckScheduler> logger,
        ILicenseManager licenseManager, ILicenseQueryEngine queryEngine, TimerExecutionOptions options)
    {
        _timerExecutor = timerExecutor;
        _logger = logger;
        _licenseManager = licenseManager;
        _queryEngine = queryEngine;
        _options = options;

        // Subscribe to timer events
        _timerExecutor.ExecutionStarted += OnTimerExecutionStarted;
        _timerExecutor.ExecutionCompleted += OnTimerExecutionCompleted;
        _timerExecutor.ExecutionError += OnTimerExecutionError;
    }

    public void Start()
    {
        try
        {
            _timerExecutor.Start(TimeSpan.FromMilliseconds(_options.DefaultIntervalMs));
            _status = SchedulerStatus.Running;
            _logger.LogInformation("License check scheduler started");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to start license check scheduler");
            _status = SchedulerStatus.Error;
            throw;
        }
    }

    public void Stop()
    {
        try
        {
            _timerExecutor.Stop();
            _status = SchedulerStatus.Stopped;
            _logger.LogInformation("License check scheduler stopped");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to stop license check scheduler");
            throw;
        }
    }

    public void Pause()
    {
        _timerExecutor.Pause();
        _status = SchedulerStatus.Paused;
        _logger.LogInformation("License check scheduler paused");
    }

    public void Resume()
    {
        _timerExecutor.Resume();
        _status = SchedulerStatus.Running;
        _logger.LogInformation("License check scheduler resumed");
    }

    public void UpdateSchedule(TimeSpan newInterval)
    {
        _timerExecutor.UpdateInterval(newInterval);
        _logger.LogInformation("License check scheduler interval updated to {Interval}", newInterval);
    }

    public SchedulerStatus Status => _status;

    private async void OnTimerExecutionStarted(object sender, TimerExecutionEventArgs e)
    {
        var args = new LicenseCheckEventArgs
        {
            ExecutionId = e.ExecutionId,
            StartTime = e.StartTime,
            CheckType = LicenseCheckType.Scheduled
        };

        LicenseCheckStarted?.Invoke(this, args);

        _logger.LogDebug("License check started: {ExecutionId}", e.ExecutionId);
    }

    private async void OnTimerExecutionCompleted(object sender, TimerExecutionEventArgs e)
    {
        try
        {
            var args = new LicenseCheckEventArgs
            {
                ExecutionId = e.ExecutionId,
                StartTime = e.StartTime,
                EndTime = e.EndTime,
                Duration = e.Duration,
                CheckType = LicenseCheckType.Scheduled,
                Success = true
            };

            // Execute license check logic
            await ExecuteLicenseCheckAsync(args);

            LicenseCheckCompleted?.Invoke(this, args);

            _logger.LogDebug("License check completed: {ExecutionId}, Duration: {Duration}ms",
                e.ExecutionId, e.Duration.TotalMilliseconds);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "License check failed: {ExecutionId}", e.ExecutionId);
        }
    }

    private async void OnTimerExecutionError(object sender, TimerErrorEventArgs e)
    {
        _status = SchedulerStatus.Error;
        _logger.LogError(e.Error, "License check scheduler error: {ConsecutiveErrors} consecutive errors",
            e.ConsecutiveErrors);
    }

    private async Task ExecuteLicenseCheckAsync(LicenseCheckEventArgs args)
    {
        try
        {
            _logger.LogDebug("Executing license check: {ExecutionId}", args.ExecutionId);

            var stopwatch = Stopwatch.StartNew();

            // Query license status
            var licenseStatus = await _queryEngine.QueryLicenseStatusAsync("localhost", 1057);

            // Process license information
            await ProcessLicenseStatusAsync(licenseStatus);

            stopwatch.Stop();

            args.LicensesChecked = licenseStatus.Features.Values.Sum(f => f.ActiveUsers.Count);
            args.FeaturesProcessed = licenseStatus.Features.Count;

            _logger.LogDebug("License check processed {Licenses} licenses across {Features} features in {Duration}ms",
                args.LicensesChecked, args.FeaturesProcessed, stopwatch.ElapsedMilliseconds);
        }
        catch (Exception ex)
        {
            args.Success = false;
            args.ErrorMessage = ex.Message;
            throw;
        }
    }

    private async Task ProcessLicenseStatusAsync(LicenseServerStatus licenseStatus)
    {
        // Process license status and identify idle licenses
        foreach (var feature in licenseStatus.Features.Values)
        {
            foreach (var license in feature.ActiveUsers)
            {
                // Check if license is idle based on usage duration
                if (license.UsageDuration.TotalMinutes > _options.IdleThresholdMinutes)
                {
                    _logger.LogDebug("Found potential idle license: {User} using {Feature} for {Duration} minutes",
                        license.UserHost, license.Feature, license.UsageDuration.TotalMinutes);

                    // TODO: Implement idle license release logic
                    // This will be implemented in a future task
                }
            }
        }
    }
}
