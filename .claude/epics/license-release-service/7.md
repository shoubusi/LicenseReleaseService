## Implementation Details

### Version Detection Implementation
```csharp
public class SolidWorksVersionDetector
{
    private static readonly Dictionary<string, string> VersionRegistryPaths = new()
    {
        ["2025"] = @"SOFTWARE\SolidWorks\SolidWorks 2025",
        ["2024"] = @"SOFTWARE\SolidWorks\SolidWorks 2024",
        ["2023"] = @"SOFTWARE\SolidWorks\SolidWorks 2023",
        ["2022"] = @"SOFTWARE\SolidWorks\SolidWorks 2022",
        ["2021"] = @"SOFTWARE\SolidWorks\SolidWorks 2021",
        ["2020"] = @"SOFTWARE\SolidWorks\SolidWorks 2020"
    };

    public async Task<List<SolidWorksVersionInfo>> DetectInstalledVersionsAsync()
    {
        var installedVersions = new List<SolidWorksVersionInfo>();

        foreach (var versionEntry in VersionRegistryPaths)
        {
            var versionInfo = await DetectSingleVersionAsync(versionEntry.Key, versionEntry.Value);
            if (versionInfo != null)
            {
                installedVersions.Add(versionInfo);
            }
        }

        return installedVersions.OrderByDescending(v => v.Version).ToList();
    }

    private async Task<SolidWorksVersionInfo> DetectSingleVersionAsync(string version, string registryPath)
    {
        try
        {
            using var key = Registry.LocalMachine.OpenSubKey(registryPath);
            if (key == null) return null;

            var installPath = key.GetValue("InstallPath") as string;
            var versionNumber = key.GetValue("Version") as string;

            if (string.IsNullOrWhiteSpace(installPath)) return null;

            return new SolidWorksVersionInfo
            {
                Version = version,
                VersionNumber = versionNumber,
                InstallPath = installPath,
                ExecutablePath = Path.Combine(installPath, "SLDWORKS.exe"),
                IsConfigured = await ValidateVersionConfigurationAsync(version, installPath)
            };
        }
        catch
        {
            return null;
        }
    }
}
```

### Version Configuration Management
```csharp
public class VersionConfigurationManager
{
    private readonly IConfiguration _configuration;
    private readonly ILogger<VersionConfigurationManager> _logger;

    public VersionConfigurationManager(IConfiguration configuration, ILogger<VersionConfigurationManager> logger)
    {
        _configuration = configuration;
        _logger = logger;
    }

    public VersionConfiguration GetVersionConfiguration(string version)
    {
        var configSection = _configuration.GetSection($"SolidWorksVersions:{version}");

        return new VersionConfiguration
        {
            Version = version,
            LicenseServer = configSection["LicenseServer"] ?? _configuration["LicenseManager:LicenseServer"],
            Port = int.Parse(configSection["Port"] ?? _configuration["LicenseManager:Port"] ?? "27000"),
            FeatureCodes = configSection.GetSection("FeatureCodes").Get<List<string>>() ?? GetDefaultFeatureCodes(version),
            Timeout = int.Parse(configSection["Timeout"] ?? _configuration["LicenseManager:Timeout"] ?? "30"),
            Priority = int.Parse(configSection["Priority"] ?? "0"),
            Enabled = bool.Parse(configSection["Enabled"] ?? "true")
        };
    }

    private List<string> GetDefaultFeatureCodes(string version)
    {
        return version switch
        {
            "2025" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            "2024" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            "2023" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            "2022" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            "2021" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            "2020" => new List<string> { "solidworks", "solidworkspremium", "solidworksprofessional" },
            _ => new List<string> { "solidworks" }
        };
    }
}
```

### Version-Specific License Query
```csharp
public class VersionSpecificLicenseQuery
{
    private readonly ProcessExecutionWrapper _processWrapper;
    private readonly VersionConfigurationManager _configManager;
    private readonly ILogger<VersionSpecificLicenseQuery> _logger;

    public async Task<LicenseQueryResult> QueryVersionLicensesAsync(string version)
    {
        var config = _configManager.GetVersionConfiguration(version);

        if (!config.Enabled)
        {
            _logger.LogInformation("Version {Version} is disabled in configuration", version);
            return new LicenseQueryResult { Version = version, IsEnabled = false };
        }

        var lmutilPath = _configuration["LicenseManager:LmutilPath"];
        var serverAddress = $"{config.LicenseServer}:{config.Port}";

        var results = new List<LicenseFeatureResult>();

        foreach (var featureCode in config.FeatureCodes)
        {
            var featureResult = await QuerySingleFeatureAsync(lmutilPath, serverAddress, featureCode, config.Timeout);
            results.Add(featureResult);
        }

        return new LicenseQueryResult
        {
            Version = version,
            IsEnabled = true,
            Features = results,
            QueryTimestamp = DateTime.UtcNow
        };
    }

    private async Task<LicenseFeatureResult> QuerySingleFeatureAsync(string lmutilPath, string serverAddress, string featureCode, int timeout)
    {
        var arguments = new List<string>
        {
            "lmstat",
            "-a",
            "-c",
            $"@{serverAddress}",
            "-f",
            featureCode
        };

        var result = await _processWrapper.ExecuteProcessAsync(lmutilPath, arguments, timeout);

        return ParseLicenseFeatureResult(featureCode, result);
    }
}
```
