## Overview

Implement robust idle detection strategies combining time-based and ping-based approaches to accurately identify inactive SolidWorks sessions, ensuring efficient license utilization while preventing false positive license releases.

## Technical Approach

### 1. Hybrid Detection Strategy
- Combine time-based inactivity detection with active ping verification
- Implement configurable detection thresholds and cooldown periods
- Support multiple detection methods for improved accuracy
- Adaptive detection based on user behavior patterns

### 2. Time-Based Detection
- Monitor system-wide user activity (keyboard, mouse, input)
- Track SolidWorks-specific activity events
- Implement graduated detection levels (warning, imminent, critical)
- Support work-hour vs. off-hour detection rules

### 3. Ping-Based Detection
- Active probing of SolidWorks processes for responsiveness
- Windows API integration for process health monitoring
- Document activity monitoring through file system events
- Network activity monitoring for collaborative sessions

### 4. State Management
- Maintain session state across detection cycles
- Implement hysteresis to prevent rapid state changes
- Support user override and manual intervention
- Persistent state recovery after service restart

### Idle Detection Engine
```csharp
public class IdleDetectionEngine
{
    private readonly IIdleDetector[] _detectors;
    private readonly IdleDetectionConfiguration _config;
    private readonly ILogger<IdleDetectionEngine> _logger;
    private readonly Timer _detectionTimer;
    private readonly Dictionary<int, SessionState> _sessionStates;

    public IdleDetectionEngine(IEnumerable<IIdleDetector> detectors,
        IdleDetectionConfiguration config,
        ILogger<IdleDetectionEngine> logger)
    {
        _detectors = detectors.ToArray();
        _config = config;
        _logger = logger;
        _sessionStates = new Dictionary<int, SessionState>();

        _detectionTimer = new Timer(DetectionCallback, null,
            TimeSpan.Zero, TimeSpan.FromMinutes(config.DetectionInterval));
    }

    private async void DetectionCallback(object state)
    {
        try
        {
            await PerformIdleDetectionAsync();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error during idle detection cycle");
        }
    }

    private async Task PerformIdleDetectionAsync()
    {
        var activeProcesses = await GetActiveSolidWorksProcessesAsync();
        var detectionResults = new Dictionary<int, IdleDetectionResult>();

        foreach (var process in activeProcesses)
        {
            var result = await DetectProcessIdleStateAsync(process);
            detectionResults[process.Id] = result;
        }

        UpdateSessionStates(detectionResults);
        RaiseIdleDetectionEvents(detectionResults);
    }

    private async Task<IdleDetectionResult> DetectProcessIdleStateAsync(Process process)
    {
        var detectorResults = new List<DetectorResult>();

        foreach (var detector in _detectors)
        {
            try
            {
                var result = await detector.DetectIdleAsync(process);
                detectorResults.Add(result);
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "Error in detector {DetectorName} for process {ProcessId}",
                    detector.GetType().Name, process.Id);
            }
        }

        return ConsolidateDetectionResults(process.Id, detectorResults);
    }
}
```

### Time-Based Detector
```csharp
public class TimeBasedIdleDetector : IIdleDetector
{
    private readonly TimeBasedDetectionConfig _config;
    private readonly ILogger<TimeBasedIdleDetector> _logger;
    private readonly Dictionary<int, DateTime> _lastActivityTimes;

    public TimeBasedIdleDetector(TimeBasedDetectionConfig config, ILogger<TimeBasedIdleDetector> logger)
    {
        _config = config;
        _logger = logger;
        _lastActivityTimes = new Dictionary<int, DateTime>();
    }

    public async Task<DetectorResult> DetectIdleAsync(Process process)
    {
        var currentTime = DateTime.UtcNow;
        var processId = process.Id;

        // Get last activity time or use process start time
        var lastActivity = GetLastActivityTime(processId, process);
        var idleDuration = currentTime - lastActivity;

        var isIdle = idleDuration.TotalMinutes > _config.IdleThresholdMinutes;
        var confidence = CalculateConfidence(idleDuration, _config.IdleThresholdMinutes);

        return new DetectorResult
        {
            DetectorType = "TimeBased",
            IsIdle = isIdle,
            Confidence = confidence,
            LastActivityTime = lastActivity,
            IdleDuration = idleDuration,
            Metadata = new Dictionary<string, object>
            {
                ["ThresholdMinutes"] = _config.IdleThresholdMinutes,
                ["CurrentIdleMinutes"] = idleDuration.TotalMinutes,
                ["ProcessStartTime"] = process.StartTime
            }
        };
    }

    private DateTime GetLastActivityTime(int processId, Process process)
    {
        // Try to get cached activity time
        if (_lastActivityTimes.TryGetValue(processId, out var cachedTime))
        {
            return cachedTime;
        }

        // Use process start time as fallback
        return process.StartTime.ToUniversalTime();
    }

    public void RecordActivity(int processId, ActivityType activityType)
    {
        _lastActivityTimes[processId] = DateTime.UtcNow;

        _logger.LogDebug("Recorded {ActivityType} activity for process {ProcessId}",
            activityType, processId);
    }
}
```

### Ping-Based Detector
```csharp
public class PingBasedIdleDetector : IIdleDetector
{
    private readonly PingBasedDetectionConfig _config;
    private readonly ILogger<PingBasedIdleDetector> _logger;

    public async Task<DetectorResult> DetectIdleAsync(Process process)
    {
        try
        {
            var pingResult = await PingProcessAsync(process);
            var documentActivity = await CheckDocumentActivityAsync(process);
            var networkActivity = await CheckNetworkActivityAsync(process);

            var isIdle = !pingResult.IsResponsive &&
                        !documentActivity.HasActivity &&
                        !networkActivity.HasActivity;

            var confidence = CalculateOverallConfidence(pingResult, documentActivity, networkActivity);

            return new DetectorResult
            {
                DetectorType = "PingBased",
                IsIdle = isIdle,
                Confidence = confidence,
                Metadata = new Dictionary<string, object>
                {
                    ["PingResponsive"] = pingResult.IsResponsive,
                    ["PingResponseTime"] = pingResult.ResponseTime,
                    ["DocumentActivity"] = documentActivity.HasActivity,
                    ["NetworkActivity"] = networkActivity.HasActivity,
                    ["LastDocumentSave"] = documentActivity.LastSaveTime,
                    ["NetworkBytesSent"] = networkActivity.BytesSent,
                    ["NetworkBytesReceived"] = networkActivity.BytesReceived
                }
            };
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error during ping-based detection for process {ProcessId}", process.Id);

            return new DetectorResult
            {
                DetectorType = "PingBased",
                IsIdle = false, // Assume not idle on error
                Confidence = 0.0,
                Error = ex.Message
            };
        }
    }

    private async Task<PingResult> PingProcessAsync(Process process)
    {
        try
        {
            // Use Windows API to check if process is responding
            var isResponding = process.Responding;

            // Additional checks for SolidWorks-specific responsiveness
            var mainWindowHandle = process.MainWindowHandle;
            var hasMainWindow = mainWindowHandle != IntPtr.Zero;

            return new PingResult
            {
                IsResponsive = isResponding && hasMainWindow,
                ResponseTime = await MeasureResponseTimeAsync(process)
            };
        }
        catch
        {
            return new PingResult { IsResponsive = false };
        }
    }

    private async Task<DocumentActivityResult> CheckDocumentActivityAsync(Process process)
    {
        try
        {
            var processPath = process.MainModule?.FileName;
            if (string.IsNullOrEmpty(processPath)) return new DocumentActivityResult();

            var directory = Path.GetDirectoryName(processPath);
            var recentFiles = GetRecentSolidWorksFiles(directory);

            return new DocumentActivityResult
            {
                HasActivity = recentFiles.Any(f =>
                    (DateTime.UtcNow - f.LastWriteTimeUtc).TotalMinutes < _config.DocumentActivityThresholdMinutes),
                LastSaveTime = recentFiles.FirstOrDefault()?.LastWriteTimeUtc,
                RecentFileCount = recentFiles.Count
            };
        }
        catch
        {
            return new DocumentActivityResult();
        }
    }
}
```

### Activity Monitoring Service
```csharp
public class ActivityMonitoringService : IDisposable
{
    private readonly IIdleDetector[] _detectors;
    private readonly System.Timers.Timer _systemActivityTimer;
    private readonly FileSystemWatcher[] _fileWatchers;
    private readonly PerformanceCounter[] _networkCounters;

    public ActivityMonitoringService(IEnumerable<IIdleDetector> detectors)
    {
        _detectors = detectors.ToArray();

        // System-wide activity monitoring
        _systemActivityTimer = new System.Timers.Timer(30000); // 30 seconds
        _systemActivityTimer.Elapsed += OnSystemActivity;

        // File system monitoring for SolidWorks directories
        _fileWatchers = SetupFileWatchers();

        // Network activity monitoring
        _networkCounters = SetupNetworkCounters();
    }

    private void OnSystemActivity(object sender, ElapsedEventArgs e)
    {
        var lastInputInfo = GetLastInputInfo();
        var timeSinceLastInput = DateTime.UtcNow - lastInputInfo.Time;

        if (timeSinceLastInput.TotalSeconds < 30) // Recent system activity
        {
            NotifyAllDetectors(ActivityType.SystemInput);
        }
    }

    private void NotifyAllDetectors(ActivityType activityType)
    {
        foreach (var timeBasedDetector in _detectors.OfType<TimeBasedIdleDetector>())
        {
            timeBasedDetector.RecordActivity(GetCurrentProcessId(), activityType);
        }
    }

    private int GetCurrentProcessId()
    {
        // This would be implemented to get the current SolidWorks process ID
        // For now, return a placeholder
        return Environment.ProcessId;
    }
}
```
